class Candidate:
    def construct(self):
        from math import isfinite as candidate_isfinite
        if self.mobjects:
            raise ValueError('candidate_requires_dedicated_empty_canvas')
        candidate_width = config.frame_width - 1
        candidate_height = config.frame_height - 1
        if not candidate_isfinite(candidate_width) or not candidate_isfinite(candidate_height) or candidate_width <= 0 or candidate_height <= 0:
            raise ValueError('candidate_invalid_viewport')
        candidate_objects = {}
        candidate_objects[(0, 1)] = MathTex('\\sum_{n=1}^{\\infty} \\frac{(-1)^n}{\\sqrt{n}}', font_size=22)
        candidate_objects[(3, 1)] = MathTex('S_n', font_size=22)
        candidate_objects[(5, 1)] = MathTex('\\sum \\frac{1}{\\sqrt{n}}', font_size=22)
        candidate_objects[(0, 0)] = Text('讨论级数 ', font='Microsoft YaHei', font_size=22)
        candidate_objects[(0, 2)] = Text(' 的敛散性，并指出是绝对收敛还是条件收敛。', font='Microsoft YaHei', font_size=22)
        candidate_objects[(1, 0)] = Text('第一步：在左侧坐标系中交替向上和向下画出正负项柱状图，展示正负交', font='Microsoft YaHei', font_size=22)
        candidate_objects[(2, 0)] = Text('替的数学形态。', font='Microsoft YaHei', font_size=22)
        candidate_objects[(3, 0)] = Text('第二步：用橙色折线连接部分和 ', font='Microsoft YaHei', font_size=22)
        candidate_objects[(3, 2)] = Text('，展示它在某一个水平线上下振荡，', font='Microsoft YaHei', font_size=22)
        candidate_objects[(4, 0)] = Text('但最终逐渐趋向一个固定值。', font='Microsoft YaHei', font_size=22)
        candidate_objects[(5, 0)] = Text('第三步：在右侧坐标系中画出绝对值级数 ', font='Microsoft YaHei', font_size=22)
        candidate_objects[(5, 2)] = Text(' 的柱状图，展示其高度递', font='Microsoft YaHei', font_size=22)
        candidate_objects[(6, 0)] = Text('减极其缓慢。', font='Microsoft YaHei', font_size=22)
        candidate_objects[(7, 0)] = Text('第四步：用折线展示绝对值级数的部分和轨迹，一直向右上方无限延伸，', font='Microsoft YaHei', font_size=22)
        candidate_objects[(8, 0)] = Text('直白地证明其发散。', font='Microsoft YaHei', font_size=22)
        candidate_objects[(9, 0)] = Text('第五步：同时展示两幅图，对比说明“原级数收敛，但绝对值级数发散，', font='Microsoft YaHei', font_size=22)
        candidate_objects[(10, 0)] = Text('因此条件收敛”。', font='Microsoft YaHei', font_size=22)
        candidate_objects[(11, 0)] = Text('条件收敛', font='Microsoft YaHei', font_size=22)
        candidate_rows = []
        candidate_rows.append(VGroup(candidate_objects[(0, 0)], candidate_objects[(0, 1)], candidate_objects[(0, 2)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(1, 0)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(2, 0)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(3, 0)], candidate_objects[(3, 1)], candidate_objects[(3, 2)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(4, 0)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(5, 0)], candidate_objects[(5, 1)], candidate_objects[(5, 2)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(6, 0)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(7, 0)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(8, 0)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(9, 0)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(10, 0)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(11, 0)]).arrange(RIGHT, buff=0))
        candidate_pages = []
        candidate_page = []
        candidate_used = 0
        for candidate_index, candidate_row in enumerate(candidate_rows):
            if not candidate_isfinite(candidate_row.width) or not candidate_isfinite(candidate_row.height) or candidate_row.width > candidate_width or candidate_row.height > candidate_height or candidate_row.height <= 0:
                raise ValueError('candidate_row_exceeds_measured_viewport')
            candidate_need = candidate_row.height + (1/4 if candidate_page else 0)
            if candidate_used + candidate_need > candidate_height:
                candidate_pages.append(candidate_page)
                candidate_page = []
                candidate_used = 0
                candidate_need = candidate_row.height
            candidate_page.append(candidate_index)
            candidate_used += candidate_need
        if candidate_page:
            candidate_pages.append(candidate_page)
        self.candidate_page_map = candidate_pages
        for candidate_indices in candidate_pages:
            candidate_group = VGroup(*[candidate_rows[i] for i in candidate_indices]).arrange(DOWN, aligned_edge=LEFT, buff=1/4)
            candidate_group.move_to(ORIGIN)
            if candidate_group.width > candidate_width or candidate_group.height > candidate_height:
                raise ValueError('candidate_page_exceeds_measured_viewport')
            self.play(FadeIn(candidate_group))
            self.wait(1)
            self.play(FadeOut(candidate_group))
            self.remove(candidate_group)
