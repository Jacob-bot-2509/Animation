class Candidate:
    def construct(self):
        from math import isfinite as candidate_isfinite
        if self.mobjects:
            raise ValueError('candidate_requires_dedicated_empty_canvas')
        candidate_width = config.frame_width - 1
        candidate_height = config.frame_height - 1
        if not candidate_isfinite(candidate_width) or not candidate_isfinite(candidate_height) or candidate_width <= 0 or candidate_height <= 0:
            raise ValueError('candidate_invalid_viewport')
        candidate_objects = {}
        candidate_objects[(0, 1)] = MathTex('\\oint_L x dy - y dx', font_size=22)
        candidate_objects[(0, 3)] = MathTex('L', font_size=22)
        candidate_objects[(0, 5)] = MathTex('x^2 + y^2 = a^2', font_size=22)
        candidate_objects[(1, 1)] = MathTex('a', font_size=22)
        candidate_objects[(2, 1)] = MathTex('L', font_size=22)
        candidate_objects[(4, 1)] = MathTex('D', font_size=22)
        candidate_objects[(4, 3)] = MathTex('L', font_size=22)
        candidate_objects[(8, 1)] = MathTex('\\frac{\\partial Q}{\\partial x} = 1', font_size=22)
        candidate_objects[(8, 3)] = MathTex('\\frac{\\partial P}{\\partial y} = -1', font_size=22)
        candidate_objects[(10, 1)] = MathTex('2\\pi a^2', font_size=22)
        candidate_objects[(11, 0)] = MathTex('2\\pi a^2', font_size=22)
        candidate_objects[(0, 0)] = Text('利用格林公式计算曲线积分 ', font='Microsoft YaHei', font_size=22)
        candidate_objects[(0, 2)] = Text('，其中 ', font='Microsoft YaHei', font_size=22)
        candidate_objects[(0, 4)] = Text(' 是正向圆周 ', font='Microsoft YaHei', font_size=22)
        candidate_objects[(0, 6)] = Text('。', font='Microsoft YaHei', font_size=22)
        candidate_objects[(1, 0)] = Text('第一步：画出直角坐标系，用红色绘制一个以原点为圆心、半径为 ', font='Microsoft YaHei', font_size=22)
        candidate_objects[(1, 2)] = Text('M', font='Microsoft YaHei', font_size=22).set_opacity(0)
        candidate_objects[(2, 0)] = Text('的圆形路径 ', font='Microsoft YaHei', font_size=22)
        candidate_objects[(2, 2)] = Text('。', font='Microsoft YaHei', font_size=22)
        candidate_objects[(3, 0)] = Text('第二步：在圆的边界上添加逆时针方向的红色箭头，标出正向路径。', font='Microsoft YaHei', font_size=22)
        candidate_objects[(4, 0)] = Text('第三步：用蓝色半透明颜色填充圆内部的区域 ', font='Microsoft YaHei', font_size=22)
        candidate_objects[(4, 2)] = Text('，代表边界 ', font='Microsoft YaHei', font_size=22)
        candidate_objects[(4, 4)] = Text(' 围成', font='Microsoft YaHei', font_size=22)
        candidate_objects[(5, 0)] = Text('的闭区域。', font='Microsoft YaHei', font_size=22)
        candidate_objects[(6, 0)] = Text('第四步：在圆的区域内随机绘制一些微小的向量箭头，示意在向量场中每', font='Microsoft YaHei', font_size=22)
        candidate_objects[(7, 0)] = Text('个点的方向。', font='Microsoft YaHei', font_size=22)
        candidate_objects[(8, 0)] = Text('第五步：画面右侧同步展示偏导数计算推导：', font='Microsoft YaHei', font_size=22)
        candidate_objects[(8, 2)] = Text('，', font='Microsoft YaHei', font_size=22)
        candidate_objects[(8, 4)] = Text('。', font='Microsoft YaHei', font_size=22)
        candidate_objects[(9, 0)] = Text('第六步：高亮显示格林公式等式，并强调左边的边界积分等价于右边区域', font='Microsoft YaHei', font_size=22)
        candidate_objects[(10, 0)] = Text('的二重积分，最终展现结果 ', font='Microsoft YaHei', font_size=22)
        candidate_objects[(10, 2)] = Text('。', font='Microsoft YaHei', font_size=22)
        candidate_rows = []
        candidate_rows.append(VGroup(candidate_objects[(0, 0)], candidate_objects[(0, 1)], candidate_objects[(0, 2)], candidate_objects[(0, 3)], candidate_objects[(0, 4)], candidate_objects[(0, 5)], candidate_objects[(0, 6)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(1, 0)], candidate_objects[(1, 1)], candidate_objects[(1, 2)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(2, 0)], candidate_objects[(2, 1)], candidate_objects[(2, 2)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(3, 0)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(4, 0)], candidate_objects[(4, 1)], candidate_objects[(4, 2)], candidate_objects[(4, 3)], candidate_objects[(4, 4)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(5, 0)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(6, 0)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(7, 0)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(8, 0)], candidate_objects[(8, 1)], candidate_objects[(8, 2)], candidate_objects[(8, 3)], candidate_objects[(8, 4)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(9, 0)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(10, 0)], candidate_objects[(10, 1)], candidate_objects[(10, 2)]).arrange(RIGHT, buff=0))
        candidate_rows.append(VGroup(candidate_objects[(11, 0)]).arrange(RIGHT, buff=0))
        candidate_pages = []
        candidate_page = []
        candidate_used = 0
        for candidate_index, candidate_row in enumerate(candidate_rows):
            if not candidate_isfinite(candidate_row.width) or not candidate_isfinite(candidate_row.height) or candidate_row.width > candidate_width or candidate_row.height > candidate_height or candidate_row.height <= 0:
                raise ValueError('candidate_row_exceeds_measured_viewport')
            candidate_need = candidate_row.height + (1/4 if candidate_page else 0)
            if candidate_used + candidate_need > candidate_height:
                candidate_pages.append(candidate_page)
                candidate_page = []
                candidate_used = 0
                candidate_need = candidate_row.height
            candidate_page.append(candidate_index)
            candidate_used += candidate_need
        if candidate_page:
            candidate_pages.append(candidate_page)
        self.candidate_page_map = candidate_pages
        for candidate_indices in candidate_pages:
            candidate_group = VGroup(*[candidate_rows[i] for i in candidate_indices]).arrange(DOWN, aligned_edge=LEFT, buff=1/4)
            candidate_group.move_to(ORIGIN)
            if candidate_group.width > candidate_width or candidate_group.height > candidate_height:
                raise ValueError('candidate_page_exceeds_measured_viewport')
            self.play(FadeIn(candidate_group))
            self.wait(1)
            self.play(FadeOut(candidate_group))
            self.remove(candidate_group)
